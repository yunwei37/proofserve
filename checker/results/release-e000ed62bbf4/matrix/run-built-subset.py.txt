#!/usr/bin/env python3
"""Prepared finite built-subset check. Invoke only after the external timing handoff."""
import sys,os,json,fcntl,datetime,importlib.util
from pathlib import Path
D=Path(__file__).resolve().parent
cfg=json.loads((D/"selection.json").read_text())
matrix=Path(cfg["matrix_raw"])
handoff=matrix/"matrix-collection-handoff.json"
matrix_end=json.loads((handoff if handoff.exists() else matrix/"phase-terminal.json").read_text())
if not (matrix_end.get("complete") or (matrix_end.get("handoff_ready") and matrix_end.get("status")=="collection_finished_with_incomplete_cells")):
 raise SystemExit("final matrix collection has not retired; no corpus process started")
# Manual invocation remains after the coordinator hands back O external timings.
# A finished collection with explicitly censored cells is not a complete matrix.
raw=D/"capture";raw.mkdir(exist_ok=True)
os.environ["PROOFSERVE_BENCH_SDK"]=cfg["sdk"]
sys.path.insert(0,str(matrix/"frozen/bench"))
import guard
spec=importlib.util.spec_from_file_location("frozen_release",matrix/"frozen/bench/release.py")
release=importlib.util.module_from_spec(spec);spec.loader.exec_module(release)
guard.ROOT=raw
roots=Path(cfg["completed_root_file"]).read_text().splitlines()
assert len(roots)==len(set(roots))==cfg["requested_built_modules"]
env=guard.environment()
original=json.loads(Path(cfg["census_run_config"]).read_text())["env"]
if "LD_LIBRARY_PATH"in original:env["LD_LIBRARY_PATH"]=original["LD_LIBRARY_PATH"]
env.update(PATH=cfg["sdk"]+"/bin:"+env["PATH"],LEAN_SYSROOT=cfg["sdk"],
 LEAN_PATH=json.loads(Path(cfg["artifact_binding"]).read_text())["evaluated_search_path"],
 LEAN_NUM_THREADS=str(cfg["native_workers"]),PROOFSERVE_BENCH_CLASSIFICATION="post-external-timing built-subset kernel check")
attempt=1
while(raw/"runs"/("built-subset-attempt-"+str(attempt))).exists():attempt+=1
key="built-subset-attempt-"+str(attempt)
remaining=(datetime.datetime(2026,11,6,20,tzinfo=datetime.timezone.utc)-datetime.datetime.now(datetime.timezone.utc)).total_seconds()
if remaining<=0:raise SystemExit("closeout time reached; no new process started")
argv=[cfg["binary"],cfg["project"],"--workers",str(cfg["native_workers"]),
 "--root-file",cfg["completed_root_file"],"--closure-roots","500",
 "--checkpoint-dir",cfg["checkpoint_dir"],"--json"]
result=None
initial=set(os.sched_getaffinity(0));previous=Path.cwd()
try:
 with Path(cfg["pcore_lease"]).open("a+")as lock:
  fcntl.flock(lock,fcntl.LOCK_EX)
  remaining=(datetime.datetime(2026,11,6,20,tzinfo=datetime.timezone.utc)-datetime.datetime.now(datetime.timezone.utc)).total_seconds()
  if remaining<=0:raise SystemExit("closeout time reached while awaiting lease")
  os.sched_setaffinity(0,set(range(24)));os.chdir(cfg["project"])
  result=guard.run(key,argv,remaining,False,env)
finally:
 os.chdir(previous);os.sched_setaffinity(0,initial)
 rows=[]if result is None else [dict(phase="whole-library",returncode=result["returncode"])]
 coverage=release.corpus_coverage(raw,roots,[],rows,bool(result and result["returncode"]==0))
 coverage.update(scope="W completed 162170-source subset",source_data_commit=cfg["source_data_commit"],
  whole_source_frame_modules=cfg["source_modules"],user_deferred_modules=cfg["deferred_modules"],
  configuration_units=cfg["configuration_units"],whole_corpus_complete=False,
  deferred_root_file=cfg["deferred_root_file"],native_commit=cfg["native_commit"])
 release.write(raw/"built-subset-coverage.json",coverage)
 print(json.dumps(coverage),flush=True)
if result is None or result["returncode"]!=0:raise SystemExit(2)
